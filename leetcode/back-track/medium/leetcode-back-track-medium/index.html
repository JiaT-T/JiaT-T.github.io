第三十九题：组合总和#

核心思路：每次添加元素之后都将target减少对应的值，直到等于或小于零

具体实现：

在回溯函数中首先判断target是否已经等于零——代表着当前元素的组合已经满足要求，可以压入res数组中

否则继续进行遍历与递归，直到出现满足总和要求的数组为止

在循环中要注意的是，当总和已经超出target时，就不要在把当前元素存入temp中了，而是直接跳过他，从下一个元素继续进行

 1void backTrack(vector<int>& candidates, int target, int index, vector<int>& temp, vector<vector<int>>& res)
 2{
 3    if(target == 0)
 4    {
 5        res.push_back(temp);
 6        return;
 7    }
 8
 9    for(int i = index; i < candidates.size(); i++)
10    {
11        if(target - candidates[i] < 0) continue;
12        temp.push_back(candidates[i]);
13        backTrack(candidates, target - candidates[i], i, temp, res);
14        temp.pop_back();
15    }
16}
17vector<vector<int>> combinationSum(vector<int>& candidates, int target)
18{
19    vector<vector<int>> res;
20    vector<int> temp;
21    res.reserve(150);
22    backTrack(candidates, target, 0, temp, res);
23    return res;
24}

第四十六题：全排列#

核心在于“交换”的步骤

首先从第一个元素开始，既然首元素已经确定了，之后就是递归地对后面的元素进行全排列

本质上就是让所有元素都当一次第一个元素，再让此时的第一个元素之后的所有元素当一次第二个元素…..以此类推，从前往后每个位置分别有n、n-1、n-2个种选择，也就是n！种排列方式

 1vector<vector<int>> permute(vector<int>& nums)
 2{
 3    vector<vector<int>> res;
 4    res.reserve(factorial(nums.size()));
 5
 6    backTrack(nums, res, 0);
 7    return res;
 8}
 9
10void backTrack(vector<int>& nums, vector<vector<int>>& res, int first)
11{
12    if(first == nums.size())
13        res.push_back(nums);
14
15    for(int i = first; i < nums.size(); i++)
16    {
17        std::swap(nums[first], nums[i]);
18        backTrack(nums, res, first + 1);
19        std::swap(nums[first], nums[i]);
20    }   
21}
22
23size_t factorial(size_t sz)
24{
25    size_t res;
26    for(int i = 0; i < sz; i++) res *= i;
27    return res;
28}

补充解法：used 数组回溯

 1class Solution
 2{
 3public:
 4    std::vector<vector<int>> res;
 5    std::vector<int> path;
 6    void backTrack(vector<int>& nums, vector<bool> used)
 7    {
 8        // 终止条件
 9        if(path.size() == nums.size())
10        {
11            res.push_back(path);
12            return;
13        }
14        
15        for(int i = 0; i < nums.size(); i++)
16        {
17            // 如果当前元素已经被使用过了
18            // 就直接跳过它
19            if(used[i])
20                continue;
21
22            // 将当前元素标记为“已使用”
23            used[i] = true;
24            path.push_back(nums[i]);
25            // 对下一个元素进行操作
26            backTrack(nums, used);
27            // 撤销所有操作
28            path.pop_back();
29            used[i] = false;
30        }
31    }
32
33    vector<vector<int>> permute(vector<int>& nums)
34    {
35        // 用来标记已使用的元素
36        std::vector<bool> used(nums.size(), false);
37        backTrack(nums, used);
38        return res;
39    }
40};

第七十八题：子集#

核心思路：枚举每一个位置，并通过递归调用进入下一个位置

具体实现：

每次进入回溯函数时，都先将当前保存的数组存入结果数组中，之后再进入具体的回溯算法——通过循环遍历当前元素之后的每一个元素，然后再递归调用相同函数，对下一个元素进行相同操作

 1void backTrack(vector<int>& nums, vector<vector<int>>& res, vector<int>& temp, int start)
 2{
 3    res.emplace_back(temp);
 4    for(int i = start; i < nums.size(); i++)
 5    {
 6        temp.push_back(nums[i]);
 7        backTrack(nums, res, temp, i + 1);
 8        temp.pop_back();
 9    }
10}
11vector<vector<int>> subsets(vector<int>& nums)
12{
13    vector<vector<int>> res;
14    vector<int> temp;
15    res.reserve(1 << nums.size());
16    backTrack(nums, res, temp, 0);
17    return res;
18}

第七十九题：单词搜索#

使用到了深搜

具体实现：

首先在双层循环中判断首元素是什么，之后以这个元素为起点进行深搜

dfs函数核心的判断逻辑是“found”变量那里——只有出现一条连续的、与word匹配的元素串，found才能为真

要注意的是，因为同一元素不能重复使用，所以在进行深搜的时候要提前将当前元素标记为“已使用”，这里采用的是‘\0’（不属于任何字母）

 1bool dfs(vector<vector<char>>& board, string word, int index, int r, int c)
 2{   
 3    if(index == word.size()) return true;
 4    if(r < 0 || r >= board.size() || c < 0 || c >= board[0].size() || word[index] != board[r][c]) return false;
 5
 6    char temp = board[r][c];
 7    board[r][c] = '\0';
 8
 9    bool found = dfs(board, word, index + 1, r + 1, c    ) || 
10                 dfs(board, word, index + 1, r,     c + 1) ||
11                 dfs(board, word, index + 1, r - 1, c    ) ||
12                 dfs(board, word, index + 1, r,     c - 1);
13
14    board[r][c] = temp;
15
16    return found;
17}
18bool exist(vector<vector<char>>& board, string word)
19{
20    for(int j = 0; j < board[0].size(); j++)
21    {
22        for(int i = 0; i < board.size(); i++)
23        {
24            if(dfs(board, word, 0, i, j)) return true;
25        }
26    }
27    return false;
28}

第一百三十一题：分割回文串#

思路：对于每一个字符，都先判断其是否能够被分割，即以start为首，以当前字符为尾的字符串是否为回文串；如果能够分割（第一条路径），就将当前元素压入path，组成子回文串，并从下一个字符开始进行新一轮的回文串判断；第二条路径为：不从当前字符进行分割，而是直接转向下一个字符，从下一个字符开始判断（为了找到更长的回文串）

 1bool isPalindrome(const string& s, int left, int right)
 2{
 3    while(left < right)
 4    {
 5        if(s[left++] != s[right--])
 6            return false;
 7    }
 8    return true;
 9}
10void dfs(string& s, int index, int start, vector<string>& path, vector<vector<string>>& res)
11{
12    const int sz = s.size();
13    if(index == sz) 
14    {
15        res.emplace_back(path);
16        return;
17    }
18
19    if(isPalindrome(s, start, index))
20    {
21        path.emplace_back(s.substr(start, index - start + 1));
22        dfs(s, index + 1, index + 1, path, res);
23        path.pop_back();
24    }
25
26    if(index < sz - 1)
27        dfs(s, index + 1, start, path, res);
28}
29
30vector<vector<string>> partition(string s)
31{
32    vector<string> path;
33    vector<vector<string>> res;
34
35    dfs(s, 0, 0, path, res);
36    return res;
37}

第十七题：电话号码的字母组合#

使用到的知识点是 哈希表 和 回溯算法

因为未知输入字符串的长度，所以不能手动的定义for循环去暴力求解

联想到树的结构，从一个分支一直走到底，再返回上一个最近的分支；与此题类似，以前一个字符串为基点，向后遍历每一个出现的字符，组成新的字符串。

难点在于如何回溯：这里使用到的是递归函数

根据传入的index确定当前数字，然后再根据数字在map中得到对应的字符，以每一个字符为起点，进行函数的递归调用，直到index与输入字符串长度相同

letterCombinations（）

 1class Solution 
 2{
 3public:
 4    void backTrack(const string& digits, int index, std::unordered_map<char, string>& phoneMap, string& combination, std::vector<string>& combinations)
 5    {
 6        if(index == digits.size()) 
 7            combinations.push_back(combination);
 8        else
 9        {
10            char num = digits[index];
11            const string& letters = phoneMap[num];
12            for(const char& letter : letters)
13            {
14                combination.push_back(letter);
15                backTrack(digits, index + 1, phoneMap, combination, combinations);
16                combination.pop_back();
17            }
18        }
19    }
20
21    vector<string> letterCombinations(string digits) 
22    {
23        if(digits.empty()) return {};
24        std::unordered_map<char, string> phoneMap
25        {
26            {'2', "abc"},
27            {'3', "def"},
28            {'4', "ghi"},
29            {'5', "jkl"},
30            {'6', "mno"},
31            {'7', "pqrs"},
32            {'8', "tuv"},
33            {'9', "wxyz"}
34        };
35        std::vector<string> combinations;
36        string combination;
37        backTrack(digits, 0, phoneMap, combination, combinations);
38        return combinations;
39    }
40};

第二十二题：括号生成#

使用到的是回溯算法

对于输入的数字n，可以知道的是最终的元素为2n个括号，其中左括号和右括号都为n个

先经过一次分析得到规律：以n=2为例，每个元素分别有2个左右括号，第一个空必须是’ （ ’，此时left的数量加一；而第二个空则有两种可能，当为‘（’时，left的数量已经达到了2，那么之后两个空就只能是‘）’；当为‘）’时，此时left = right，第三个空就只能是‘（’，从而得出第四个空是‘）’

总结规律，可以得出：

1.当left < n时，应该添加‘（’

2.当right < left时，应当有‘）’与其配对，故添加‘）’

在此条件下进行递归，最后当left+right与n相同时终止最后一层递归，返回上一层的枝，继续下一个子节点的递归

generateParenthesis()

 1class Solution
 2{
 3public:
 4    vector<string> generateParenthesis(int n)
 5    {
 6        std::vector<string> combinations;
 7        string combination;
 8        backTrace(0, 0, combination, combinations, n);
 9        return combinations;
10    }
11
12private :
13    void backTrace(int left, int right, string& combination, std::vector<string>& combinations, int n)
14    {
15        if(left + right == 2 * n) 
16        {
17            combinations.push_back(combination);
18            return;
19        }
20        else
21        {
22            if(left < n) 
23            {
24                combination.push_back('(');
25                backTrace(left + 1, right, combination, combinations, n);
26                combination.pop_back();
27            }
28            if(right < left)
29            {
30                combination.push_back(')');
31                backTrace(left, right + 1, combination, combinations, n);
32                combination.pop_back();
33            }
34        }
35    }
36};
